import { SEOEngineClient } from '@seoengine/node-sdk';

// Initialize technical SEO client
const client = new SEOEngineClient({
  apiKey: process.env.SEO_ENGINE_KEY,
});

const execution = await client.crawls.trigger({
  projectId: 'prj_core_01',
  crawlType: 'deep_render',
  maxDepth: 5,
  dispatchWebhooks: true,
});

console.log(`Job scheduled: ${execution.id}`);
